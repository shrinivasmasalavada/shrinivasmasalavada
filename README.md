    const time =
        clock.getElapsedTime();


    /* Drone floating */

    drone.position.y =
        Math.sin(time * 1.4) * .25;


    /* Drone rotation */

    drone.rotation.y += .003;


    drone.rotation.x =
        -0.12 +
        mouseY * .15;


    drone.rotation.z =
        -0.08 +
        mouseX * .15;


    /* Propellers */

    propellers.forEach(
        (prop,index) => {

            prop.rotation.y +=
                index % 2 === 0
                ? .35
                : -.35;

        }
    );


    /* Particles */

    particles.rotation.y =
        time * .015;


    particles.rotation.x =
        Math.sin(time * .1) * .05;


    /* Mouse camera */

    camera.position.x +=
        (mouseX * .8 - camera.position.x)
        * .03;

    camera.position.y +=
        (2.2 - mouseY * .5 - camera.position.y)
        * .03;

    camera.lookAt(
        0,
        0,
        0
    );


    renderer.render(
        scene,
        camera
    );

}


animate();

        height,
        false
    );

}
